Código Google Apps Script (codigo.gs) 

// CÓDIGO PRODUCCIÓN GOOGLE APPS SCRIPT PARA BDXX_HUARMEY
// Pegar este código en Extensiones > Apps Script de su Google Sheet BDXX_HUARMEY

function doGet(e) {
  var action = e ? e.parameter.action : '';
  if (action === 'getMaestros') {
    return getMaestros();
  }
  return ContentService.createTextOutput(JSON.stringify({ status: 'success', message: 'API Movilidades Huarmey Activa' }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    if (data.action === 'syncRegistros') {
      return syncRegistros(data.records);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: 'Acción no reconocida' }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function getMaestros() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('maestros');
  if (!sheet) {
    return responseJSON({ status: 'error', message: 'No se encontró la hoja maestros' });
  }
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    return responseJSON({ status: 'success', data: [] });
  }
  
  var headers = data[0].map(function(h){ return String(h).trim().toLowerCase(); });
  var rows = [];
  
  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var item = {};
    for (var j = 0; j < headers.length; j++) {
      item[headers[j]] = row[j];
    }
    rows.push(item);
  }
  
  return responseJSON({ status: 'success', data: rows });
}

function syncRegistros(records) {
  if (!records || !records.length) {
    return responseJSON({ status: 'success', syncedIds: [] });
  }
  
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName('registros');
  if (!sheet) {
    return responseJSON({ status: 'error', message: 'No se encontró la hoja registros' });
  }
  
  var syncedIds = [];
  
  for (var k = 0; k < records.length; k++) {
    var r = records[k];
    sheet.appendRow([
      r.fecha || new Date(),
      r.placa || '',
      r.conductor || '',
      r.tipo || '',
      r.ruta || '',
      r.cultivo || '',
      r.capacidad || 0,
      r.pasajeros || 0,
      r.libres || 0,
      r.observaciones || ''
    ]);
    syncedIds.push(r.id);
  }
  
  return responseJSON({ status: 'success', syncedIds: syncedIds });
}

function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

Instrucciones de Despliegue en Google Sheets:

  1. Cree el archivo Google Sheet con nombre BDXX_HUARMEY.
  2. Cree las pestañas maestros y registros.
  3. En maestros agregue encabezados: placa, conductor, tipo, capacidad, ruta, cultivo, CELULAR, ESTADO.
  4. En registros agregue encabezados: fecha, placa, conductor, tipo, ruta, cultivo, capacidad, pasajeros, libres, observaciones.
  5. Vaya a Extensiones > Apps Script, pegue el código anterior y haga clic en Implementar > Nueva implementación.
  6. Seleccione tipo: Aplicación Web, Acceso: Cualquier persona (Anyone).
  7. Copie la URL generada y péguela arriba en esta pantalla.
